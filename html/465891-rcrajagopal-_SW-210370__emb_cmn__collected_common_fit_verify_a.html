+#define FIT_MAX_KEY_NAME		64
+#define PREFIX_KEY			"key-"

+static const uint8_t *fit_get_node_hash(const void *fit, int node)
+{
+	int hash;
+	const char *prop;
+
+	hash = fdt_subnode_offset(fit, node, HL_FIT_HASH_PROP);
+	if (hash < 0) {
+		EMB_LOG_ERR("Failed to find hash node");
+		return NULL;
+	}
+
+	prop = (const char *)fdt_getprop(fit, hash, HL_FIT_ALGO_PROP, NULL);
+	if (strcmp(prop, HL_FIT_SEC_SHA_NAME)) {
+		EMB_LOG_RAW_ERR("Hash Algorithm is not matching, ");
+		EMB_LOG_RAW("algo = ");
+		EMB_LOG_RAW(prop);
+		EMB_LOG_RAW("\n\r");
+		return NULL;
+	}
+
+	return (const uint8_t *)fdt_getprop(fit, hash, HL_FIT_VALUE_PROP, NULL);
+}

+__weak void sha_csum(const unsigned char *input, unsigned int ilen, unsigned char *output) {}
+
+int fit_verify_image_hash(const void *fit, int node, const uint8_t *buf, int len, void *hash)
+{
+	const uint8_t *node_hash;
+	uint8_t calc_hash[SHA_SUM_LEN];
+
+	node_hash = fit_get_node_hash(fit, node);
+	if (!node_hash) {
+		EMB_LOG_ERR("Failed at fetching the hash node");
+		return FIT_FORMAT_NO_PROPERTY;
+	}
+
+	sha_csum(buf, len, calc_hash);
+	if (memcmp(node_hash, calc_hash, SHA_SUM_LEN)) {
+		EMB_LOG_ERR("Hash verification failed for the image");
+		return FIT_SEC_HASH_FAILED;
+	}
+
+	if (hash)
+		memcpy(hash, calc_hash, SHA_SUM_LEN);
+
+	return FIT_SEC_NONE;
+}

+static int fit_get_pub_key_node(const void *pub_key_fit, const char *key_name)
+{
+	int sig_node;
+	char key_node_name[FIT_MAX_KEY_NAME] = PREFIX_KEY;
+	size_t key_node_name_len = sizeof(PREFIX_KEY) - 1;
+
+	key_node_name[FIT_MAX_KEY_NAME - 1] = '\0';
+
+	strncpy(key_node_name + key_node_name_len,
+			key_name, FIT_MAX_KEY_NAME - key_node_name_len);
+	/*
+	 * To avoid stack based buffer overflow
+	 * make sure your key string length does not exceed buffer size.
+	 */
+	if (key_node_name[FIT_MAX_KEY_NAME - 1] != '\0')
+		return -1;
+
+	sig_node = fdt_subnode_offset(pub_key_fit, 0, HL_FIT_SIGNATURE_NODENAME);
+
+	return fdt_subnode_offset(pub_key_fit, sig_node, key_node_name);
+}

+int fit_get_key_prop(const void *pub_key_fit, const char *key_name,
+		     struct rsa_key_prop *prop)
+{
+	int pub_key_node = 0;
+	int length;
+
+	if (fdt_check_header(pub_key_fit)) {
+		EMB_LOG_ERR("FITkey header check failed");
+		return FIT_FORMAT_HEADER_CHECK_FAILED;
+	}
+
+	pub_key_node = fit_get_pub_key_node(pub_key_fit, key_name);
+	if (pub_key_node < 0) {
+		EMB_LOG_ERR("Wrong FITkey format: no key node");
+		return pub_key_node;
+	}
+
+	prop->num_bits =
+		fit_get_int(pub_key_fit, pub_key_node, "rsa,num-bits", 0);
+	prop->n0inv =
+		fit_get_int(pub_key_fit, pub_key_node, "rsa,n0-inverse", 0);
+	prop->public_exponent =
+		fdt_getprop(pub_key_fit, pub_key_node, "rsa,exponent", &length);
+	if (!prop->public_exponent || length < sizeof(uint64_t))
+		prop->public_exponent = NULL;
+	prop->modulus =
+		fdt_getprop(pub_key_fit, pub_key_node, "rsa,modulus", NULL);
+	prop->rr =
+		fdt_getprop(pub_key_fit, pub_key_node, "rsa,r-squared", NULL);
+
+	if (!prop->num_bits || !prop->modulus) {
+		EMB_LOG_ERR("FIT: missing RSA Key");
+		return FIT_FORMAT_NO_PROPERTY;
+	}
+
+	return FIT_FORMAT_SUCCESS;
+}

+static int fit_verify_img_sig(const void *img, size_t img_size, const void *sec_desc,
+			      size_t sec_desc_size, const void *pub_key_fit,
+			      const uint8_t *sig, size_t sig_len)
+{
+	const char *key_name = HL_FIT_SEC_KEY_NAME;
+	struct rsa_key_prop prop;
+	struct image_region region[HL_SEC_REGIONS_CNT];
+
+	if (fit_get_key_prop(pub_key_fit, key_name, &prop))
+		return 1;
+
+	/*
+	 * region 0 holds the fit image as a whole,
+	 * region 1 holds the hl security descriptor
+	 */
+	region[0].data = img;
+	region[0].size = img_size;
+	region[1].data = sec_desc;
+	region[1].size = sec_desc_size;
+
+	return rsa_verify_regions(region, HL_SEC_REGIONS_CNT, sig, sig_len, &prop);
+}

+int fit_verify_security_footer(const void *img, size_t img_size, enum fit_sec_failure *res)
+{
+	const void *fit_sec_off;
+	const void *pub_key_fit;
+
+	/* hl fit files are padded so the size is 4 bytes aligned */
+	img_size = ROUND_UP4(img_size);
+
+	if (!security_img_validation_is_enabled())
+		return 0;
+
+	/* security descriptor is at the footer of the image */
+	fit_sec_off = (const void *)((uint8_t *)img + img_size);
+
+	/* check that the calculated memory range are legal */
+	if (!security_sram_is_valid_address_range((uintptr_t)fit_sec_off,
+						 sizeof(fit_sec))) {
+		EMB_LOG_ERR("Invalid address range");
+		return 1;
+	}
+
+	memcpy(&fit_sec, fit_sec_off, FIT_SEC_DESC_SIZE);
+
+	if (fit_sec.magic_number != FIT_SEC_MAGIC) {
+		EMB_LOG_ERR("Bad magic, security descriptor not found");
+		return 1;
+	}
+
+	if ((fit_sec.desc_version_maj != FIT_SEC_DESC_VER_MAJOR) ||
+	    (fit_sec.desc_version_min != FIT_SEC_DESC_VER_MINOR)) {
+		EMB_LOG_ERR("security descriptor version mismatch");
+		return 1;
+	}
+
+	if (fit_sec.desc_size < FIT_SEC_DESC_SIZE) {
+		EMB_LOG_ERR("security descriptor size too small");
+		return 1;
+	}
+
+	if (fit_sec.img_size != img_size) {
+		EMB_LOG_ERR("image size mismatch with security descriptor");
+		return 1;
+	}
+
+	if (!security_validate_svn(fit_sec.sec_version)) {
+		EMB_LOG_ERR("security descriptor svn not valid");
+		*res = FIT_SEC_SVN_INVALID;
+		return 1;
+	}
+
+	if (security_validate_key_idx(fit_sec.key_num)) {
+		EMB_LOG_HEX_ERR("Key revoked: ", fit_sec.key_num);
+		*res = FIT_SEC_KEY_REVOKED;
+		return 1;
+	}
+
+	pub_key_fit = &fit_sec.pub_key;
+
+	if (!security_validate_pub_key(fit_sec.key_num, pub_key_fit,
+				       FIT_PUB_KEY_SIZE, NULL, 0)) {
+		EMB_LOG_ERR("Public Key validation failed");
+		*res = FIT_SEC_SIGN_FAILED;
+		return 1;
+	}
+
+	/*
+	 * signature is calculated on both the fit image and the security
+	 * descriptor with the signature field of the descriptor zeroized.
+	 */
+	memcpy(signature, fit_sec.signature, FIT_SIGNATURE_SIZE);
+	memset(fit_sec.signature, 0, FIT_SIGNATURE_SIZE);
+
+	if (fit_verify_img_sig(img, img_size, &fit_sec, FIT_SEC_DESC_SIZE,
+			       pub_key_fit, signature, FIT_SIGNATURE_SIZE)) {
+		EMB_LOG_ERR("image signature verification failed");
+		*res = FIT_SEC_SIGN_FAILED;
+		return 1;
+	}
+
+	security_store_svn_into_fw_info(fit_sec.sec_version);
+	security_store_key_index_into_fw_info(fit_sec.key_num);
+	security_store_key_index_into_boot_info(fit_sec.key_num);
+
+	return 0;
+}

+static bool fit_load_is_secured(const void *fit)
+{
+	int err;
+	uint32_t load_type;
+	struct fit_error fit_err;
+	bool is_secured = false;
+	size_t fit_size = fdt_totalsize(fit);
+
+	err = fit_get_load_type(fit, &load_type, &fit_err);
+	if (err) {
+		is_secured = true;
+		goto tpm_pcr_extend;
+	}
+
+	switch (load_type) {
+	case FIT_LOAD_TYPE_RUN:
+	case FIT_LOAD_TYPE_RUN_N_EXTEND:
+	case FIT_LOAD_TYPE_NIC_EN_MASK:
+		is_secured = true;
+		break;
+	default:
+		is_secured = false;
+		break;
+	}
+
+tpm_pcr_extend:
+	if (is_secured)
+		if (security_is_tpm_ext_req(load_type))
+			security_ext_tpm_pcr(fit, fit_size);
+
+	return is_secured;
+}

+int fit_verify(void *fit, struct fit_error *err)
+{
+	int node, ret;
+	int conf, num_images, i;
+	const u8 *buf;
+	size_t fit_size = fdt_totalsize(fit);
+	size_t len;
+
+	err->fmt = fit_check_format(fit);
+	if (err->fmt) {
+		err->load = FIT_LOAD_CORRUPTED;
+		return err->load;
+	}
+
+	/* Check if image supports a minimum FDT version or newer */
+	if (fdt_version(fit) < FDT_MIN_SUPPORTED_VERSION) {
+		err->load = FIT_LOAD_SEC_FAILED;
+		err->sec = FIT_SEC_NONE;
+		return err->sec;
+	}
+
+	/* check provided fit_size is inside the bounds */
+	if (!security_is_valid_address_range((uintptr_t)fit, fit_size)) {
+		err->load = FIT_LOAD_SEC_FAILED;
+		err->sec = FIT_SEC_NONE;
+		return 1;
+	}
+
+	/* Get number of images in preboot configuration */
+	ret = fit_get_conf_offset_and_num_of_loadables(fit, &conf, &num_images);
+	if (ret || (conf < 0)) {
+		err->load = FIT_LOAD_CORRUPTED;
+		err->sec = FIT_SEC_NONE;
+		return 1;
+	}
+
+	if (num_images > FIT_MAX_IMAGES) {
+		EMB_LOG_HEX_ERR("Images in fit exceeds max limit: ", FIT_MAX_IMAGES);
+		err->load = FIT_LOAD_CORRUPTED;
+		return 1;
+	} else if (num_images < 0) {
+		EMB_LOG_HEX_ERR("Cant read images from fit: ", -num_images);
+		err->load = FIT_LOAD_CORRUPTED;
+		return 1;
+	}
+
+	if (!fit_load_is_secured(fit))
+		return 0; /* Nothing more to check */
+
+	if (fit_verify_security_footer(fit, fit_size, &err->sec)) {
+		err->load = FIT_LOAD_SEC_FAILED;
+		return 1;
+	}
+
