+
+def my_func(x):
+    default_stream = ht.hpu.default_stream()
+    y = x.add(1.0)
+    
+    ht.hpu.set_stream(s0)
+    z = y.add(2.0)
+    z = z.mul(2.0)
+
+    ht.hpu.set_stream(s1)
+    w = y.add(2.0)
+    w = w.mul(2.0)
+    
+    ht.hpu.set_stream(default_stream)
+    s0.synchronize()
+    s1.synchronize()
+    
+    res = z + w
+    res = res.t()
+    
+    return res
+
+lop.start() 
+
+lop.emit_begin(lop.profiler.INIT)
+in0 = torch.rand(16, 16).to("hpu")
+lop.emit_end(lop.profiler.INIT)
+
+lop.emit_begin(lop.profiler.LOOP)
+for i in range(50):
+  lop.emit_begin(lop.profiler.ITERATION)
+  res = my_func(in0)
+  lop.emit_end(lop.profiler.ITERATION)
+lop.emit_end(lop.profiler.LOOP)
+
+lop.stop()
+lop.flush()

+void ProfilerEngine::copying_handler(ProfilerEngine* engine) {

+                volatile auto end = ring->e;

+                if (begin < end) {
+                    // Single memcopy can handle it.
+                    auto num_of_elements = end - begin;

+                    if (engine->events_counter + num_of_elements > EVENT_TABLE_SIZE) {
+                        printf("Reached event limit. Bailing out.\n");
+                        limit_reached = true;
+                        break;
+                    }
+
+                    memcpy(engine->events_table + engine->events_counter, &ring->events[begin], num_of_elements * sizeof(Event));
+
+                    engine->events_counter += num_of_elements;
+                    ring->b = end;
+
+                    if (engine->biggest_copy < num_of_elements) engine->biggest_copy = num_of_elements;
+
+                } else if (begin > end) {
+                    // Two memcopies needed.
+                    auto num_of_elements1 = LOP_RING_SIZE - begin;
+                    auto num_of_elements2 = end;
+
+                    if (engine->events_counter + num_of_elements1 + num_of_elements2 > EVENT_TABLE_SIZE) {
+                        printf("Reached event limit. Bailing out.\n");
+                        limit_reached = true;
+                        break;
+                    }
+
+                    memcpy(engine->events_table + engine->events_counter, &ring->events[begin], num_of_elements1 * sizeof(Event));
+                    memcpy(engine->events_table + engine->events_counter + num_of_elements1, &ring->events[0], num_of_elements2 * sizeof(Event));
+
+                    engine->events_counter += num_of_elements1 + num_of_elements2;
