[PNP-202] api: update arcpid api to version 2.0.0

The ARCPID API provides a structured interface for monitoring and
controlling power and thermal management functionality. The
implementation includes both a firmware component (C-based) and a
user-space interactive shell (Python-based).

Version 2.0.0 of the ARCPID API represents a significant architectural
redesign from the previous implementation.

This upgrade was necessary for several critical reasons:

* Improved Scalability: The previous API architecture lacked a scalable
  design pattern, which created growing pains as the system expanded
* Simplified Configuration: Each API command used to be configured both
  in the firmware and in user shell, creating consistency issues. The
  new API uses a centralized JSON file.
* Enhanced Functionality: Command hierarchy, custom commands, error
  handling and more
* Unifying API into a single repository: API files were split between
  the ARCPID FW repo and SiVal repo, making it difficult to maintain and
  update. The new implementation consolidates all API files into the FW
  repository.

New FW architecture:
* Command format: <block> <param> [<option>] <command> [<data>]
* Define commands in a JSON file
* Commands header is auto-generated using a Python script

New User Shell architecture:
* Python-based CLI with tab completion and command history
* Support for new commands is added automatically (using JSON parsing)
* Version checks: comparing FW API version to parsed JSON version
* Multiple output formats including state, decimal, hex, and tuples

Add all ARCPID API V2 related files to ARCPID repo, replacing the old
implementation.

+class HlthunkControl:
+    """Python wrapper class for the hlthunk library with focus on control functions"""
+
+    def __init__(self, device_id: int = 0, busid: Optional[str] = None, lib_path: str = "libhl-thunk.so"):
+        """
+        Initialize hlthunk control interface
+
+        Args:
+            device_id: The device ID number of the control device as appears in /dev/
+            busid: PCI address of the device on the host PCI bus (optional)
+            lib_path: Path to the hlthunk library
+
+        Returns:
+            0 for success, negative value for error
+        """
+        self.fd = -1  # Initialize file descriptor to invalid
+
+        # Load the library
+        try:
+            self.lib = ctypes.CDLL(lib_path)
+        except OSError as e:
+            print(f"Error: Failed to load libhl-thunk library: {e}")
+            return
+
+        # Set function argument and return types
+        self._setup_function_signatures()
+
+        # Convert busid to bytes if provided
+        busid_bytes = busid.encode('utf-8') if busid else None
+        busid_ptr = ctypes.c_char_p(busid_bytes) if busid_bytes else None
+
+        # Open control device
+        self.fd = self.lib.hlthunk_open_control(device_id, busid_ptr)
+        if self.fd < 0:
+            print(f"Error: Failed to open hlthunk device (device_id={device_id}, busid={busid}), error code: {self.fd}")
+            if self.fd == -errno.ENOENT:
+                print("Check that the device driver is loaded and operational")
+            elif self.fd == -errno.EACCES:
+                print("Check that you have permission to access the device")
+
+    def _setup_function_signatures(self):
+        """Set up C function signatures for the library functions we'll use"""
+
+        # hlthunk_close
+        self.lib.hlthunk_close.argtypes = [ctypes.c_int]
+        self.lib.hlthunk_close.restype = ctypes.c_int
+
+        # hlthunk_fw_send_generic_request
+        self.lib.hlthunk_fw_send_generic_request.argtypes = [
+            ctypes.c_int,       # fd
+            ctypes.c_void_p,    # buff
+            ctypes.c_uint32,    # buff_size
+            ctypes.c_uint32     # sub_opcode
+        ]
+        self.lib.hlthunk_fw_send_generic_request.restype = ctypes.c_int
+
+        # hlthunk_open_control (already used in __init__)
+        self.lib.hlthunk_open_control.argtypes = [ctypes.c_int, ctypes.c_char_p]
+        self.lib.hlthunk_open_control.restype = ctypes.c_int
+
+        # Add signature for hlthunk_get_pci_bus_id_from_fd
+        self.lib.hlthunk_get_pci_bus_id_from_fd.argtypes = [
+            ctypes.c_int,       # fd
+            ctypes.c_char_p,    # pci_bus_id
+            ctypes.c_int        # len
+        ]
+        self.lib.hlthunk_get_pci_bus_id_from_fd.restype = ctypes.c_int
+
+    def close(self) -> int:
+        """
+        Close the device
+
+        Returns:
+            int: Return value of the close() syscall
+                 0 for success, negative value for error
+        """
+        if hasattr(self, 'fd') and self.fd >= 0:
+            result = self.lib.hlthunk_close(self.fd)
+            self.fd = -1
+            return result
+        return 0
+
+    def fw_send_generic_request(self, buffer: int, sub_opcode: int) -> Tuple[int, int]:
+        """
+        Send a generic request to the firmware using a 64-bit integer buffer
+
+        Args:
+            buffer: 64-bit integer value to be sent to the FW, may also serve as output value
+            sub_opcode: Generic request sub-opcode
+
+        Returns:
+            Tuple[int, int]: (return code, output buffer)
+                             return code 0 for success, negative value for error
+        """
+        if not hasattr(self, 'fd') or self.fd < 0:
+            print("Error: Device is not open")
+            return -1, 0
+
+        # Create a 64-bit integer buffer
+        c_buffer = ctypes.c_uint64(buffer)
+        buff_size = ctypes.sizeof(c_buffer)
+
+        # Call the function
+        result = self.lib.hlthunk_fw_send_generic_request(
+            self.fd,
+            ctypes.byref(c_buffer),
+            buff_size,
+            sub_opcode
+        )
+
+        # Return result and possibly modified buffer value
+        if result != 0:
+            print(f"Error: fw_send_generic_request failed with error code: {result}")
+
+        return result, c_buffer.value
+
+    def get_pci_bus_id(self) -> Tuple[int, str]:
+        """
+        Get the PCI bus ID of the device
+
+        Returns:
+            Tuple[int, str]: (return code, PCI bus ID string)
+                             return code 0 for success, negative value for error
+                             PCI bus ID in format [domain]:[bus]:[device].[function]
+        """
+        if not hasattr(self, 'fd') or self.fd < 0:
+            print("Error: Device is not open")
+            return -1, ""
+
+        # PCI bus ID is typically up to 13 characters (including null terminator)
+        # Format: [domain]:[bus]:[device].[function]
+        # Example: 0000:3b:00.0
+        buf_size = 32  # Using a bigger buffer to be safe
+        pci_bus_id = ctypes.create_string_buffer(buf_size)
+
+        result = self.lib.hlthunk_get_pci_bus_id_from_fd(self.fd, pci_bus_id, buf_size)
+
+        if result != 0:
+            print(f"Error: Failed to get PCI bus ID, error code: {result}")
+            return result, ""
+
+        # Convert bytes to string and remove null characters
+        return 0, pci_bus_id.value.decode('utf-8')

+class ApiCommandProcessor:

+    def _load_json_file(self, file_path: str) -> Dict[str, Any]:

+        try:

+    def _process_commands_and_options(self, block_name: str, param_name: str) -> Tuple[Dict, Dict]:
+        """Process commands and options for a parameter."""
+        result = {}
+
+        # Process commands
+        commands_dict = {}
+        for cmd in self.json_data["blocks"][block_name]["params"][param_name]["commands"]:
+            # First check in command-specific section if available
+            if param_name in self.json_data["commands"]:
+                if cmd in self.json_data["commands"][param_name]:
+                    commands_dict[cmd] = {
+                        "id": self.json_data["commands"][param_name][cmd]["id"],
+                        "description": self.json_data["commands"][param_name][cmd]["description"]
+                    }
+                    continue
+
+            # Then check in global commands
+            if cmd in self.json_data["commands"]["global"]:
+                commands_dict[cmd] = {
+                    "id": self.json_data["commands"]["global"][cmd]["id"],
+                    "description": self.json_data["commands"]["global"][cmd]["description"]
+                }
+
+        # Process options
+        options_dict = {}
+        options_list = self.json_data["blocks"][block_name]["params"][param_name]["options"]
+
+        # Find the right section for options
+        for opt_section, opt_values in self.json_data["options"].items():
+            for opt in options_list:
+                if opt in opt_values:
+                    options_dict[opt] = {
+                        "id": opt_values[opt]["id"],
+                        "description": opt_values[opt]["description"]
+                    }
+
+        return commands_dict, options_dict
+
+    def _build_nested_dictionary(self) -> Dict[str, Any]:
+        """Build the nested dictionary structure from the JSON data."""
+        result = {}
+
+        # First level: blocks
+        for block_name, block_data in self.json_data["blocks"].items():
+            result[block_name] = {
+                "id": block_data["id"],
+                "description": block_data["description"],
+                "params": {}
+            }
+
+            # Second level: params
+            for param_name, param_data in block_data["params"].items():
+                commands_dict, options_dict = self._process_commands_and_options(block_name, param_name)
+
+                result[block_name]["params"][param_name] = {
+                    "id": param_data["id"],
+                    "description": param_data["description"],
+                    "commands": commands_dict,
+                    "options": options_dict
+                }
+
+        return result
+
+    def dump_nested_dict(self):
+        """
+        Dump the nested dictionary.
+        """
+        print(json.dumps(self.nested_dict, indent=2))
+
+    def create_nested_completer(self) -> NestedCompleter:
+        """
+        Create a nested completer structure for prompt_toolkit based on the parsed dictionary.
+
+        Returns:
+            NestedCompleter: A completer for use with prompt_toolkit
+        """
+        completer_dict = {}
+
+        # First level: blocks
+        for block_name, block_data in self.nested_dict.items():
+            completer_dict[block_name] = {}
+
+            # Second level: params
+            for param_name, param_data in block_data["params"].items():
+                completer_dict[block_name][param_name] = {}
+
+                # If options exist, add them as the third level
+                if param_data["options"]:
+                    for option_name in param_data["options"]:
+                        completer_dict[block_name][param_name][option_name] = {}
+
+                        # Add commands as the fourth level (for params with options)
+                        for cmd_name in param_data["commands"]:
+                            completer_dict[block_name][param_name][option_name][cmd_name] = None
+                else:
+                    # Add commands directly as the third level (for params without options)
+                    for cmd_name in param_data["commands"]:
+                        completer_dict[block_name][param_name][cmd_name] = None
+
+        return NestedCompleter.from_nested_dict(completer_dict)
