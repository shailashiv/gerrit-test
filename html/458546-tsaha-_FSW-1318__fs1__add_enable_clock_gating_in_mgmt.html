+static void fs1_clock_gating_init(void *masks, enum fs1_die_id die_id)
+{
+	const struct device *cgm_dev = CLK_GATING_DEV_INST;
+	struct fs_clock_gating_info cgm_info = {
+			.die_id = die_id,
+			.redundancy_ctx = masks,
+	};
+
+	if (cgm_dev == NULL) {
+		LOG_ERR("Clock gating driver not found, skipping init");
+		return;
+	}
+
+	clk_gating_inject_info(cgm_dev, (void *)&cgm_info);
+	clk_gating_enable_all(cgm_dev, true);

+__weak bool fs1_engine_init(struct hl_dev_config *dev_config)
+{
+	const struct device *eng_binning_dev = ENGINE_BINNING_DEV_INST;
+	enum fs1_die_id die_id = ((CFG_RREG32(mmIO_0_CFG_HALF_0_PSOC_ARC_MNG_LOCAL_BASE +
+			mmARC_LOCAL_CONF_DIE_CPU_ID) & ARC_LOCAL_CONF_DIE_CPU_ID_DIE_ID_M) >>
+			ARC_LOCAL_CONF_DIE_CPU_ID_DIE_ID_S);
+	struct fs1_binning_info info;
+
+	if (die_id <= FS1_COMPUTE_DIE_3)
+		masks = (void *)k_calloc(1, sizeof(struct fs1_redundancy_ctx_compute));
+	else if (die_id <= FS1_IO_DIE_1_HALF_1)
+		masks = (void *)k_calloc(1, sizeof(struct fs1_redundancy_ctx_io));
+	else {
+		LOG_ERR("%s: invalid die id = %u", __func__, die_id);
+		return -1;
+	}
+
+	if (masks == NULL) {
+		LOG_ERR("%s Memory not allocated.\n", __func__);
+		return -1;
+	}
+
+	info.redundancy_ctx = masks;
+	fs1_binning_drv_get_binning_info(eng_binning_dev, &info);
+
+	fs1_clock_gating_init(masks, die_id);
+	return true;
+}
