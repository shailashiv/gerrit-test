+# Function to find outliers based on IQR
+def find_outliers(data):
+    Q1 = np.percentile(data, 25)
+    Q3 = np.percentile(data, 75)
+    IQR = Q3 - Q1
+    lower_fence = Q1 - 1.5 * IQR
+    upper_fence = Q3 + 1.5 * IQR
+    outliers = data[(data < lower_fence) | (data > upper_fence)]
+    return outliers
