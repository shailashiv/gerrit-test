+#define MAX_SIZE_NAME 20
+
+typedef enum {
+	INTR_MSG_REI_DERR,
+	INTR_MSG_REI_SERR,
+	INTR_MSG_SEI,
+	INTR_MSG_SPI_ECO,
+	INTR_MSG_DPI,
+	INTR_MSG_TS,
+	INTR_MSG_RESERVED,
+	INTR_MSG_MAX
+} intr_msg_type_t;
+
+typedef struct {
+	uint32_t address;
+	uint32_t value;
+} reg_info_t;
+
+typedef struct {
+	reg_info_t m2w;  /* Message to Wire register info */
+	reg_info_t w2m;  /* Wire to Message register info */
+	reg_info_t ip;   /* Interrupt Pending register info */
+} reg_clr_info_t; //intr_sts_info_t
+
+typedef struct {
+	reg_info_t inj;        /* Injection register data */
+	uint32_t timestamp;    /* Time of injection. After a certain timeout, debug registers will be read
+						   and the test will be marked as failed */
+} intr_inj_t;
+
+typedef struct {
+	uint32_t isr_hit_cnt;  /* Number of times an ISR is hit. This is to know if there was flooding */
+} intr_det_t;
+
+typedef struct {
+	uint32_t arc_index;    /* Architecture index */
+	uint32_t aggr_index;   /* Aggregator index */
+	intr_inj_t injection;  /* Injection information */
+	intr_det_t detection;  /* Detection information */
+	reg_clr_info_t debug_info; /* Debug information */
+} intr_info_t;
+
+typedef struct {
+	intr_info_t *intr_list;  /* List of interrupt information */
+	uint32_t intr_count;     /* Number of interrupts */
+	bool result;             /* Result of the interrupt message */
+} intr_msg_t;
+
+typedef enum {
+	COMPONENT_CPU = 0,
+	COMPONENT_D2D_MAC,
+	COMPONENT_D2D_PHY,
+	COMPONENT_DEC,
+	COMPONENT_DTSTDVM,
+	COMPONENT_ENC,
+	COMPONENT_MUC,
+	COMPONENT_NIC,
+	COMPONENT_NMS,
+	COMPONENT_NOC,
+	COMPONENT_PCIE,
+	COMPONENT_PLL,
+	COMPONENT_PMMU,
+	COMPONENT_PSOC,
+	COMPONENT_RESERVED,
+	COMPONENT_XHA,
+	COMPONENT_MAX,
+} component_id_t;
+
+typedef struct {
+	char component_name[MAX_SIZE_NAME];  /* Name of the component */
+	component_id_t component_id;         /* ID of the component */
+	intr_msg_t intr_msgs[INTR_MSG_MAX];  /* Interrupt messages for the component */
+	bool result;                         /* Result of the component */
+} component_t;
+
+typedef struct {
+	component_t components[COMPONENT_MAX];  /* List of components */
+	bool result;                            /* Result of the component list */
+} component_list_t;
+
+typedef enum {
+	SINGLE,    /* Single firing mode */
+	FULL_AUTO, /* Full automatic firing mode */
+} intr_firing_mode;
+
+typedef struct {

+	intr_firing_mode firing_mode;     /* Firing mode of the interrupts */
+} sys_intr_info_t;

+static int get_component_id_by_name(const char *component_name, uint32_t *component_id)
+{
+	if (strcmp(component_name, "CPU") == 0) {
+		*component_id = COMPONENT_CPU;
+	} else if (strcmp(component_name, "D2D_MAC") == 0) {
+		*component_id = COMPONENT_D2D_MAC;
+	} else if (strcmp(component_name, "D2D_PHY") == 0) {
+		*component_id = COMPONENT_D2D_PHY;
+	} else if (strcmp(component_name, "DEC") == 0) {
+		*component_id = COMPONENT_DEC;
+	} else if (strcmp(component_name, "DTSTDVM") == 0) {
+		*component_id = COMPONENT_DTSTDVM;
+	} else if (strcmp(component_name, "ENC") == 0) {
+		*component_id = COMPONENT_ENC;
+	} else if (strcmp(component_name, "MUC") == 0) {
+		*component_id = COMPONENT_MUC;
+	} else if (strcmp(component_name, "NIC") == 0) {
+		*component_id = COMPONENT_NIC;
+	} else if (strcmp(component_name, "NMS") == 0) {
+		*component_id = COMPONENT_NMS;
+	} else if (strcmp(component_name, "NOC") == 0) {
+		*component_id = COMPONENT_NOC;
+	} else if (strcmp(component_name, "PCIE") == 0) {
+		*component_id = COMPONENT_PCIE;
+	} else if (strcmp(component_name, "PLL") == 0) {
+		*component_id = COMPONENT_PLL;
+	} else if (strcmp(component_name, "PMMU") == 0) {
+		*component_id = COMPONENT_PMMU;
+	} else if (strcmp(component_name, "PSOC") == 0) {
+		*component_id = COMPONENT_PSOC;
+	} else if (strcmp(component_name, "XHA") == 0) {
+		*component_id = COMPONENT_XHA;
+	} else {
+		printk("Invalid Component name %s\n", component_name);
+		return -1;
+	}
+
+	return 0;
+}

+static int get_debug_info(reg_clr_info_t *debug_info)
+{
+	/*TODO: can we derive the address and from aggregator index */
+	debug_info->ip.value = CFG_RREG32(debug_info->ip.address);
+	if (debug_info->ip.value) {
+		printk("ip status address: 0x%x and its value: 0x%x\n", debug_info->ip.address,  debug_info->ip.value);
+	} else {
+		printk("ip status is not set\n");
+		return -1;
+	}
+
+	debug_info->w2m.value = CFG_RREG32(debug_info->w2m.address);
+	if (debug_info->w2m.value) {
+		printk("w2m status address: 0x%x and its value: 0x%x\n", debug_info->w2m.address,  debug_info->w2m.value);
+	} else {
+		printk("w2m status is not set\n");
+		return -1;
+	}
+
+	debug_info->m2w.value = CFG_RREG32(debug_info->m2w.address);
+	if (debug_info->m2w.value) {
+		printk("m2w status address: 0x%x and its value: 0x%x\n", debug_info->m2w.address,  debug_info->m2w.value);
+	} else {
+		printk("m2w status is not set\n");
+		return -1;
+	}
+
+	return 0;
+}

+void main(void)
+{
+	uint32_t component_id;
+	component_t component;
+	uint32_t intr_msg_type;
+	bool is_single_mode = false;
+
+	char mode[10];
+	char comp_in[10];
+	char intr_type_in[10];
+
+	uint32_t inj_addr;
+
+	while (1) {
+		printk("\nEnter command: ");
+		scanf("%9s %9s %9s", mode, comp_in, intr_type_in);
+		for (int i = 0; mode[i]; i++) {
+			mode[i] = toupper(mode[i]);
+		}
+		for (int i = 0; comp_in[i]; i++) {
+			comp_in[i] = toupper(comp_in[i]);
+		}
+		for (int i = 0; intr_type_in[i]; i++) {
+			intr_type_in[i] = toupper(intr_type_in[i]);
+		}
+
+		if (strcmp(mode, "FULL") == 0) {
+			printk("Full mode selected\n");
+			component_id = COMPONENT_CPU;
+			intr_msg_type = INTR_MSG_REI_DERR;
+		} else if (strcmp(mode, "SINGLE") == 0) {
+			printk("Single mode selected\n");
+			if (get_component_id_by_name(comp_in, &component_id)) {
+				printk("Invalid Component name %s\n", comp_in);
+				continue;
+			}
+
+			if (get_intr_msg_type(intr_type_in, &intr_msg_type)) {
+				printk("Invalid Interrupt type %s\n", intr_type_in);
+				continue;
+			}
+		
+			is_single_mode = true;
+		} else {
+			printk("Invalid mode %s\n", mode);
+			continue;
+		}
+
+		/* TODO: Check whether do we need to enable the interrupts */
+		for (int i = component_id; i < COMPONENT_MAX; i++) {
+			component = io_die_comp[i];
+			for (int j = intr_msg_type; j < INTR_MSG_MAX; j++) {
+				printk("Component: %s, Interrupt Type: %s\n", comp_in, intr_type_in);
+				if (component.intr_msgs[j].intr_count) {
+					for (int k = 0; k < component.intr_msgs[j].intr_count; k++) {
+						inj_addr = component.intr_msgs[j].intr_list[k].injection.inj.address;
+						if (inj_addr) {
+							printk("ARC interrupt index: %d\n", component.intr_msgs[j].intr_list[k].arc_index);
+							printk("Aggregator interrupt index: %d\n", component.intr_msgs[j].intr_list[k].aggr_index);
+							CFG_WRREG32(inj_addr, component.intr_msgs[j].intr_list[k].injection.inj.value);
+							component.intr_msgs[j].intr_list[k].injection.timestamp = k_uptime_get();
+							if (!(get_debug_info(&component.intr_msgs[j].intr_list[k].debug_info))) {
+								component.intr_msgs[j].result = true;
+							} else {
+								component.intr_msgs[j].result = false;
+							}
+						} else {
+							printk("Interupt injection mechanism not available\n");
+						}
+					}
+				} else {
+					printk("%s interrupt not present\n", intr_type_in);
+				}
+			}
+
+			if (is_single_mode)
+				goto finish;
+		}
+
+finish:
+		printk("Interrupts fired\n");
+	}
+
