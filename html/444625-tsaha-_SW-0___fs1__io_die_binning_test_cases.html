+

+	k_usleep(200);

+	/* Half: 0, write predefined values to same test register of each decoder and read back*/

+	for (dec_index = 0 ; dec_index < 9; dec_index++) {

+		dec_addr = mmIO_0_CFG_HALF_0_MEDIA_DEC_0_VCMD_BASE + consecutive_dec_offset * dec_index;

+		wr_val = wr_val + 1;

+		CFG_WREG32(dec_addr + DEC_TEST_REGISTER, wr_val);

+		k_usleep(100);

+		rd_val = CFG_RREG32(dec_addr + DEC_TEST_REGISTER);

+		LOG_INF("Tamal %s: wr_val = 0x%08x, rd_val = 0x%08x", __func__, wr_val, rd_val);

+	}

+void io_die_binning_validation_step_1(void)

+{

+	/* ******* step 1 ***********

+	 * Efuse DEC, NIC, SKU values for Palladium validation: START

+	 * select faulty dec, nic for test purpose

+	 */

+	sys_write32(((1 << FS1_FAULTY_VDEC_5)       | /* ( IO DIE 0 - Half die 0: Dec */

+				(1 << (FS1_FAULTY_VDEC_4 + 9))  | /* ( IO DIE 0 - Half die 1: Dec */

+				(1 << (FS1_FAULTY_NIC_4 + 19))),  /* ( IO DIE 0 - Nic */

+				(0xf9c92080 + 0x4c));             /* DeC and NIC */

+

+	sys_write32(0x1000000, (0xf9c92080 + 0xf0)); /* SKU: IO die Stand Alone Mode */

+

+	/* ******** step 2 ********

+	 * pick some register and write some different value to

+	 * every engine

+	 */

+	configure_same_test_register_of_each_decoder();

+	configure_same_test_register_of_each_nic();

+	

+}
