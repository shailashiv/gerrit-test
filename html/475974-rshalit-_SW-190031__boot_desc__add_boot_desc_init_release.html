[SW-190031] boot_desc: add boot_desc_init/release

As part of moving duplicated boot_desc implementation
and variations from the applications and unifying
it, this patch adds the boot_desc_init/release
code to the cmn-lib projects, so that they can be used
as generic function for all projects.
Also, it adds free/alloc callbacks which are used
in the generic code.

+static enum boot_ptrs_type last_ptrs_initialized = BOOT_PTRS_MAX_NUM;
+
+void register_boot_desc(struct boot_desc_func *func)
+{
+	boot_desc_func.alloc_desc_offset = func->alloc_desc_offset;
+	boot_desc_func.alloc_image_desc = func->alloc_image_desc;
+	boot_desc_func.free_desc_offs = func->free_desc_offs;
+	boot_desc_func.free_image_desc = func->free_image_desc;
+}
+
+void boot_desc_release(void)
+{
+	if (!img_descs)
+		return;
+
+	boot_desc_func.free_image_desc(img_descs);
+	last_ptrs_initialized = BOOT_PTRS_MAX_NUM;
+	img_descs = NULL;
+}
+
+int boot_desc_init(enum boot_ptrs_type ptrs_type)
+{
+	unsigned int count, i;
+	u32 ptr_base, spi_off;
+	u32 *desc_offs;
+	u8 *descs_ptr;
+
+	if (ptrs_type == last_ptrs_initialized)
+		return boot_ptrs.count;
+
+	ptr_base = (ptrs_type == BOOT_PTRS_FW) ? HL_FW_BOOT_PTRS_BASE : HL_EROM_BOOT_PTRS_BASE;
+
+	/* Read the main boot pointers structure */
+	spi_flash_copy(&boot_ptrs, ptr_base, BOOT_PTRS_SZ, "BOOT PTRS");
+	if (boot_ptrs.version > 0) {
+		if (boot_ptrs.boot_ptrs_sz > BOOT_PTRS_SZ)
+			EMB_LOG_WRN("boot_ptrs size is bigger than expected");
+	}
+	boot_desc_print_magic("FW boot PTRs", &boot_ptrs.magic);
+
+	if (boot_ptrs.magic.val != HL_FW_BOOT_MAGIC)
+		goto err_exit;
+
+	count = boot_ptrs.count;
+
+	desc_offs = boot_desc_func.alloc_desc_offset(count);
+	if (!desc_offs)
+		goto err_exit;
+
+	if (is_legacy_boot_descs(&boot_ptrs)) {
+		/*
+		 * In legacy boot ptrs meta data structure the last member is "count", the legacy
+		 * offsets array starts where the "pub_key_sz" member is.
+		 */
+		spi_off = ptr_base + offsetof(struct hl_fw_boot_ptrs, pub_key_sz);
+		/* legacy descriptors don't have size field. so constant value is used */
+		boot_ptrs.img_desc_sz = LEGACY_IMAGE_DESCRIPTOR_SIZE;
+	} else {
+		spi_off = ptr_base + boot_ptrs.offset2array;
+	}
+
+	spi_flash_copy(desc_offs, spi_off, sizeof(u32) * count, "DESC OFFS");
+
+	img_descs = boot_desc_func.alloc_image_desc((count * boot_ptrs.img_desc_sz), desc_offs);
+	if (!img_descs)
+		goto err_exit;
+
+	/* Read all descriptors */
+	for (i = 0; i < count; i++) {
+		spi_off = ptr_base + desc_offs[i];
+		descs_ptr = img_descs + i * boot_ptrs.img_desc_sz;
+		spi_flash_copy(descs_ptr, spi_off, boot_ptrs.img_desc_sz, "IMG DESC");
+		if (!boot_desc_img_desc_is_valid((struct hl_fw_img_desc *)descs_ptr)) {
+			EMB_LOG_HEX_ERR("Bad boot descriptor: %d", i);
+			boot_desc_func.free_desc_offs(desc_offs);
+			goto err_exit;
+		}
+	}
+
+	boot_desc_func.free_desc_offs(desc_offs);
+	last_ptrs_initialized = ptrs_type;
+
+	return count;
+
+err_exit:
+	boot_desc_release();
+	memset(&boot_ptrs, 0, sizeof(boot_ptrs));
+	last_ptrs_initialized = BOOT_PTRS_MAX_NUM;
+
+	return 0;
+}
