+    def convert_string_to_hex(self, pci_addr, field_name, input_string):
+        # pci_addr = self.devices[device_idx]['short_pci_address']
+        log_manager.log_browser(self.pci_to_mezz_log_format[pci_addr])
+        try:
+            input_string = str(input_string)

+                return f'0x{input_string[2:].lstrip("0") or "0"}'
+            elif input_string.startswith("0b"):
+                return hex(int(input_string, 2))
+            elif input_string.isdigit():
+                return hex(int(input_string))
+            else:
+                return '0x'+''.join([hex(ord(char))[2:] for char in input_string])
+        except Exception as exception:
+            hlsa_logger(level='error', ident='Description', test_code=self.test_code,
+                        operation=f'Exception during converting ({field_name}) from string to decimal',
+                        read_value=input_string, dev_address=pci_addr)
