+static void fs_fw_isr(const struct device *dev)
+{
+	const struct fs_fw_config *config = DEV_CFG(dev);
+	struct fs_fw_data *data = DEV_DATA(dev);
+	uint32_t cause = (config->cfg_base + mmPSOC_FW_MAILBOX_INTR_SPI_CAUSE_MAILBOX_WR);
+	uint32_t status = sys_read32((mem_addr_t)(cause)) & BIT(config->mbx_idx);
+	uint32_t value = sys_read32((mem_addr_t)config->cfg_base);
+
+	if (!status) {
+		LOG_DBG("Interrupt does not belong to dev %p", dev);
+		return;
+	}
+
+	if (data->callback)
+		data->callback(dev, config->mbx_idx, value);
+	sys_write32(BIT(config->mbx_idx), (mem_addr_t)cause);
+}
+
+static void fs_fw_register_callback(const struct device *dev, fs_fw_callback_t cb)
+{
+	struct fs_fw_data *data = dev->data;
+
+	data->callback = cb;
+}
+
+static int fs_fw_irq_init(const struct device *dev)
+{
+	const struct fs_fw_config *config = DEV_CFG(dev);
+	struct fs_fw_data *const data = DEV_DATA(dev);
+	uint32_t val = sys_read32((mem_addr_t) config->intr_mask);
+
+	data->shared_irq_dev = device_get_binding(DT_LABEL(DT_NODELABEL(sw_malibox_irq)));
+	if (!data->shared_irq_dev) {
+		LOG_ERR("Failed to get sw_malibox_irq device binding");
+		return -ENODEV;
+	}
+
+	/* Register for interrupt */
+	shared_irq_isr_register(data->shared_irq_dev, (isr_t)fs_fw_isr, dev);
+	/* Unmask the psoc interrupt */
+	val &= ~BIT(config->mask_idx);
+	sys_write32(val, config->intr_mask);
+	/* Enable the psoc interrupt */
+	shared_irq_enable(data->shared_irq_dev, dev);
+
+	LOG_INF("FW Mbx idx %u", config->mbx_idx);
+	return 0;
+}
+
+static int fs_fw_init(const struct device *dev)
+{
+	LOG_INF("FS FW mbx comms dev %p enumeration start", dev);
+
+	fs_fw_irq_init(dev);
+
+	LOG_INF("FS FW mbx comms dev %p enumeration end", dev);
+
+	return 0;
+}

+ * These callbacks execute in interrupt context. Therefore, use only
+ * interrupt-safe APIS. Registration of callbacks is done via
+ * @a fs_fw_mbx_register_callback_t
+ *
+ * @param dev Driver instance
+ * @param mbx_idx Fw mbx index.
+ * @param data Message data pointer. The correct amount of data to read out
+ *        must be inferred using the message id/upper level protocol.
+ */
+typedef void (*fs_fw_callback_t)(const struct device *dev, uint32_t mbx_idx, uint32_t data);
+
+/**
+ * @typedef fs_fw_mbx_register_callback_t
+ * @brief Callback API upon registration
+ *
+ * See @a fs_fw_mbx_register_callback() for argument definitions.
+ */
+typedef void (*fs_fw_mbx_register_callback_t)(const struct device *port, fs_fw_callback_t cb);
+
+__subsystem struct fs_fw_driver_api {
+	fs_fw_mbx_register_callback_t register_callback;
+};
+
+/**
+ * @brief Register a callback function for incoming messages.
+ *
+ * @param dev Driver instance pointer.
+ * @param cb Callback function to execute on incoming message interrupts.
+ * @param user_data Application-specific data pointer which will be passed
+ *        to the callback function when executed.
+ */
+__syscall void fs_fw_mbx_register_callback(const struct device *dev, fs_fw_callback_t cb);

+static inline void z_impl_fs_fw_mbx_register_callback(const struct device *dev, fs_fw_callback_t cb)
+{
+	const struct fs_fw_driver_api *api =
+		(const struct fs_fw_driver_api *)dev->api;
