+void hl_print_sram_ecc_status(void)
+{
+	int dcore, sram, wrap;
+	u32 offset, val, addr;
+	u32 num_mem_wrapper = regmap_readl(mmDCORE0_SRAM0_BANK_SPECIAL_MEM_NUMOF);
+
+	for (dcore = 0; dcore < NUM_OF_DCORES ; dcore++) {
+		offset = dcore * GAUDI2_DCORE_OFFSET;
+		for (sram = 0; sram < NUM_OF_SRAM_BANK_PER_DCORE; sram++) {
+			for (wrap = 0; wrap < num_mem_wrapper; wrap++) {
+				regmap_writel(wrap,
+					      mmDCORE0_SRAM0_BANK_SPECIAL_MEM_ECC_SEL + offset);
+				val = regmap_readl(mmDCORE0_SRAM0_BANK_SPECIAL_MEM_ECC_ERR_STS + offset);
+				if (val) {
+					addr = regmap_readl(mmDCORE0_SRAM0_BANK_SPECIAL_MEM_ECC_ERR_ADDR + offset);
+					preboot_puts_hex64("sram ecc at address ", addr, false);
+					preboot_puts_hex("wrap ", wrap, false);
+					preboot_puts_hex("err status ", val, false);
+				}
+			}
+			offset += GAUDI2_SRAM_BANK_OFFSET;
+		}
+	}
+}
