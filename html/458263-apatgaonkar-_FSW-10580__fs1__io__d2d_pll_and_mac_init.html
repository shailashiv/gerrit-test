+static int boot_rom_d2d_pll_init(void)
+{
+	uint64_t start;
+	uint32_t dphy0_sts, dphy1_sts, dphy2_sts, dphy3_sts;

+	/* Init PLL on all DPHYS */

+	regmap_writel(0x1, IO0_H0_SOC_0_ID2D_PHY_1_CTL0_BASE(mmCOMP_DPHY_CTRL_PLL_SEQ_CTRL));
+	regmap_writel(0x1, IO0_H1_SOC_1_ID2D_PHY_0_CTL0_BASE(mmCOMP_DPHY_CTRL_PLL_SEQ_CTRL));
+	regmap_writel(0x1, IO0_H1_SOC_1_ID2D_PHY_1_CTL0_BASE(mmCOMP_DPHY_CTRL_PLL_SEQ_CTRL));
+
+	start = boot_rom_get_ts_us();
+	while (1) {
+		dphy0_sts = regmap_readl(IO0_H0_SOC_0_ID2D_PHY_0_CTL0_BASE

+		dphy1_sts = regmap_readl(IO0_H0_SOC_0_ID2D_PHY_1_CTL0_BASE
+				(mmCOMP_DPHY_CTRL_PLL_SEQ_DONE));
