+def custom_all_reduce_sum(tensor_in, buffer_in, world_size, num_shards=1):
+    """
+    Performs a custom all-reduce operation on the input tensor by splitting it into
+    shards, distributing chunks of each shard across processes using all-to-all communication,
+    performing a local reduction on the received data, and then gathering the reduced
+    chunks back across all processes.
+
+    This function is particularly useful for scenarios where all-reduce operations are
+    required in FP8 precision on HPUs that do not allreduce FP  operations.
+    By splitting the tensor into smaller shards, this method
+    enables pipelining of the summation (reduction) operation alongside the all-gather
+    communication, which can help improve performance by overlapping computation and
+    communication.
+
+    The `num_shards` parameter controls the number of tensor splits (shards) used, allowing
+    fine-tuning to achieve optimal performance. Increasing the number of shards can improve
+    pipelining efficiency by reducing the time spent waiting for data transfers, but may
+    increase overhead if too many shards are used. Experiment with different `num_shards`
+    values to find the ideal balance between performance and overhead for your specific
+    workload.
+
+    Args:
+        tensor_in (torch.Tensor): The input tensor to be reduced across processes.
+                                  This tensor will be split into shards for distributed processing.
+        buffer_in (torch.Tensor): A buffer tensor of the same size as `tensor_in`
+                                  used to hold intermediate results during the all-to-all communication.
+        world_size (int): The total number of processes participating in the all-reduce operation.
+                          This should match the world size used when initializing the distributed
+                          process group. The world size is used to ensure that tensors are divided
+                          and distributed correctly across all processes.
+        num_shards (int): The number of shards to split `tensor_in` into for processing.
+                          Each shard will be processed independently. Changing the shard
+                          count can enable better pipelining of reduction and gathering. Default is 1(No Sharding).
+    Returns:
+        torch.Tensor: The fully reduced tensor, with contributions from all processes.
+                      The reduction is applied across all processes, and each process
+                      ends up with the same fully reduced tensor.
+
+    Example Usage:
+        >>> import torch.distributed as dist
+        >>> tensor_in = torch.ones(8)
+        >>> buffer_in = torch.zeros_like(tensor_in)
+        >>> dist.init_process_group(backend='hccl', world_size=4, rank=0)
+        >>> result = custom_all_reduce(tensor_in, buffer_in, num_shards=2)
+
+    Method:
+        - Split `tensor_in` into `num_shards` for distributed processing.
+        - For each shard:
+            - Extract a portion of `buffer_in` and `tensor_in`.
+            - Scatter the shard across all processes using `all_to_all_single`.
+            - Perform a local reduction (sum) on the received data.
+        - Gather all reduced data across processes to produce the final reduced tensor.
+
+    Note:
+        - The input tensor dimensions should be compatible with the world size (number of processes).
+        - The function performs an in-place reduction on `tensor_in`, modifying its contents.
+    """
+
+    shard_size = tensor_in.size(0) // num_shards
+    start_offset = 0
+
+    reduced_chunk_list = []  # List to store reduced chunks for all_gather later
+    tensor_list = []
+
+    # Process each shard in sequence
+    for i in range(num_shards):
+        # Determine the current shard size (handles the remainder for the last shard)
+        curr_shard_size = shard_size if i < num_shards - 1 else tensor_in.size(0) - start_offset
+        buffer = buffer_in[start_offset : start_offset + curr_shard_size]
+        tensor = tensor_in[start_offset : start_offset + shard_size]
+        tensor_list.append(tensor)
+        # Update offset for next iteration
+        start_offset += curr_shard_size
+
+        # Perform all_to_all_single communication (scatter across all processes)
+        dist.all_to_all_single(buffer, tensor)
+
+        # Reshape to prepare for local reduction (summing over each received chunk)
+        buffer = buffer.view(world_size, -1)
+
+        # Perform local reduction (sum along the first dimension)
+        reduced_chunk = buffer.sum(dim=0)
+        reduced_chunk_list.append(reduced_chunk)
+
+    # Perform all_gather to collect the reduced chunks from all processes
+    work_list = []
+    for tensor, reduced_chunk in zip(tensor_list, reduced_chunk_list):
+        w = dist.all_gather_into_tensor(tensor, reduced_chunk, async_op=True)
+        work_list.append(w)
+    for w in work_list:
+        w.wait()
+
+    return tensor_in  # The fully reduced and gathered tensor
