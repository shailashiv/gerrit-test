+function swap_file {
+	local INF=$1
+	local OUTF=$2
+	local DBG=${3:-}
+
+	# SWAP needs this to be 4B aligned in size
+	dd status=none conv=sync bs=4 if=${INF} > ${OUTF}.tmp
+	# Sadly --endian=big is not swapping for -tx1 so I used some gawk magic
+	od -Anone -v -tx1 -w4 ${OUTF}.tmp | gawk '{ print $4 $3 $2 $1 }' | xxd -r -p > ${OUTF}
+
+	if [ -z "$DBG" ]; then
+		rm -f ${OUTF}.tmp
+	fi
