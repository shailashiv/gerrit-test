+DO io_d0_mgmt_arc_64bit.cmm
+SYStem.Mode.Up
+Core 0
+
+;Provide the binary path
+&BINPATH="~/trees/npu-stack/embedded/buildroot-external/output/fs1_arc/images/boot_rom.bin"
+&ELFPATH="~/trees/npu-stack/embedded/buildroot-external/output/fs1_arc/build/arcboot-custom/boot_rom"
+
+OPEN #1 "&BINPATH" /Read /Binary
+&bin_sz=OS.FILE.SIZE("&BINPATH")
+GLOBAL &VLD0 &VLD1 &INC &ITER &SCRUB &PROG &DATA
+
+&ITER=&bin_sz/4
+
+&SCRUB=1; Set to scrub boot rom
+&PROG=1; Set to program boot rom
+&LOAD_ELF=1; Set to load ELF
+
+IF &SCRUB==1
+(
+  GOSUB prepare
+  GOSUB scrub
+)
+
+IF &PROG==1
+(
+  GOSUB prepare
+  GOSUB prog
+  PRINT "BOOT ROM PROG DONE"
+)
+
+IF &LOAD_ELF==1
+(
+Data.Load.Elf &ELFPATH /NoCODE
+)
+
+
+END
+
+prepare:
+  &VLD0=0x40000000
+  &VLD1=0xC0000000
+  &INC=0
+RETURN
+
+scrub:
+RePeaT &ITER
+(
+  &DATA=0xAABBCCDD
+  GOSUB data_tx
+)
+RETURN
+
+prog:
+RePeaT &ITER
+(
+  PRIVATE &byte0 &byte1 &byte2 &byte3
+  READ #1 &byte0
+  READ #1 &byte1
+  READ #1 &byte2
+  READ #1 &byte3
+
+  &DATA=&byte0|&byte1<<8.|&byte2<<16.|&byte3<<24.
+  GOSUB data_tx
+)
+CLOSE #1
+RETURN
+
+data_tx:
+  &VLD0=&VLD0+&INC
+  PRINT "&VLD0"
+  &VLD1=&VLD1+&INC
+  PRINT "&VLD1"
+  D.S enaxi:0x300007FF9CD4E0C %LE %Long &DATA
+  D.S enaxi:0x300007FF9CD4E08 %LE %Long &VLD0
+  D.S enaxi:0x300007FF9CD4E08 %LE %Long &VLD1
+  D.S enaxi:0x300007FF9CD4E08 %LE %Long &VLD0
+  &INC=1
+RETURN

+
+	&SPI_BYTE_SWAP=0xF9CB03A0	;H15_PROJECT ➔ IO_0_CFG ➔ HALF_0 ➔ PSOC ➔ PSOC_CFG ➔ GLOBAL_CONF ➔ SPI_DIRECT_WR_RD_CTRL
+	&SPI_WRITE_LIMIT=0xF9CB03F0	;H15_PROJECT ➔ IO_0_CFG ➔ HALF_0 ➔ PSOC ➔ PSOC_CFG ➔ GLOBAL_CONF ➔ SPI_WRITE_LIMIT
+	&SPI_BASE=0xF9CAF000		;H15_PROJECT ➔ IO_0_CFG ➔ HALF_0 ➔ PSOC ➔ PSOC_CFG ➔ QSPI
+	&RAM_BASE=0
+
+	; Enable Write till 16MB
+	Data.Set A:&SPI_WRITE_LIMIT %LE %Long 0x5		;	TODO: value same as in FW//1 << (16+5) = 2MB
+
+	; Init SPI controller
+	Data.Set A:&SPI_BASE+0x08 %LE %Long 0x0			; SSIENR: what is SSI? DW_apb_ssi
+	Data.Set A:&SPI_BASE+0x00 %LE %Long 0x700C0		; CTRLR0: controls the serial data transfer.
+	Data.Set A:&SPI_BASE+0x04 %LE %Long 0x0			; CTRLR1: This register exists only when the DW_apb_ssi is configured as a master device.
+	Data.Set A:&SPI_BASE+0x14 %LE %Long 0x2			; BAUDR: SSI Clock Divider. Fssi_clk/Div
+	Data.Set A:&SPI_BASE+0x10 %LE %Long 0x0			; SER
+	Data.Set A:&SPI_BASE+0xF0 %LE %Long 0x1			; RX_SAMPLE_DLY - Shift delay
+	Data.Set A:&SPI_BASE+0x08 %LE %Long 0x1
+
+	;---------------------- LOAD SPI FLASH ----------------
+	Break.RESet
+	FLASHFILE.RESet
+	FLASHFILE.CONFIG  &SPI_BASE           0
+	//FLASHFILE.TARGET <<code range>>       <<data range>>                  <<algorithm file>>
+	FLASHFILE.TARGET &RAM_BASE++0x2FFF    (&RAM_BASE+0x3000)++0x27FF     ./spi64_dwssi.hs.le.bin  /keep   /stacksize 0x400
+	FLASHFILE.GETID
+	FLASHFILE.unlock 0x0--0xFFFFFF
+	FLASHFILE.erase 0x0--0xFFFFFF
+	; FLASHFILE.lock 0x0--0xFFFF
+	Data.LOAD.Binary &SPIFLASH_BINPATH 0x94000000
+)

+

+SYStem.Option IMASKHLL ON   ; Disable interrupts while stepping over high level langage  lines(C code)

+    ENDDO
+  )
