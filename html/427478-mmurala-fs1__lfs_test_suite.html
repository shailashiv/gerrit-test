+int boot_info_get_size(__unused char *tag, __unused enum hl_boot_info_type file_num, uint32_t *size)
+{
+	static char full_name[LFS_NAME_MAX] = { '\0' };
+	lfs_file_t file;
+	int ret;
+
+	ret = lfs_init_disk();
+	if (ret)
+		return ret;
+
+	lfs_get_file_name(full_name, tag, file_num);
+	ret = lfs_file_opencfg(&lfs, &file, full_name, LFS_O_RDONLY, &defaults);
+	if (ret < 0) {
+		*size = 0;
+		return 0;
+	}
+
+	*size = (uint32_t)lfs_file_size(&lfs, &file);
+
+	lfs_file_close(&lfs, &file);
+
+	return ret;
+}
+/* Modify data inside an existing info. Same constraints as lfs_file_set() */
+int lfs_file_set_partial(__unused char *tag, __unused enum hl_boot_info_type file_num, const void *data, uint32_t offset, uint32_t size)
+{
+	static char full_name[LFS_NAME_MAX] = { '\0' };
+	lfs_file_t file;
+	int ret;
+	size_t fsize;
+
+	ret = lfs_init_disk();
+	if (ret)
+		return ret;
+
+	lfs_get_file_name(full_name, tag, file_num);
+	ret = lfs_file_opencfg(&lfs, &file, full_name, LFS_O_RDWR | LFS_O_CREAT, &defaults);
+	if (ret < 0)
+		return ret;
+
+	fsize = lfs_file_size(&lfs, &file);
+	if (fsize > size + offset) {
+		ret = lfs_file_truncate(&lfs, &file, size + offset);
+	}
+
+	if (!ret) {
+		lfs_file_seek(&lfs, &file, offset, LFS_SEEK_SET);
+		ret = lfs_file_write(&lfs, &file, data, size);
+	}
+
+	lfs_file_close(&lfs, &file);
+
+	return ret >= 0 ? 0 : ret;
+}
