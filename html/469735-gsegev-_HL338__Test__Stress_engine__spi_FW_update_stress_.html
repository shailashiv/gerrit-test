+"""
+Functionality: Shuffle loading between 2 FW images for a predetermined number of loops, counting number of failures
+               and successes
+Input: self
+Output: Print to log number of failures and successes
+PRECONDITIONS: prior running the test the tester need to update the fw packages versions test_options and add the fw
+packages in the relevant directories for the test as in test_options. 2 fw packages for the testing and another
+latest_fw package for loading the machine after the test is done. if there aren't dedicated fw packages directories, 
+the test will create them in its first run but the tester will have to copy inside the relevant fw packages and
+rerun the test
+"""
+
+import hw_manager
+import setup_manager
+import pytest
+from tests.test_base import TestBase
+from setup_manager.hlsa_logger import hlsa_logger, hlsa_checker
+
+log_manager = setup_manager.options.get_log_manager()
+
+
+class TestSpiFwUpdateLoops(TestBase):
+
+    fw_pack_location = "/software/releases/%s/firmware/%s"
+
+    @classmethod
+    def setup_class(cls):
+        try:
+            # Declare global boolean result variable
+            cls.global_status = True
+
+            # Init devices dict to hold the under-test devices
+            cls.devices = {}
+
+            # Get test_options
+            hlsa_logger(level='debug', ident='Description', test_code='HL338_FW_UPDATE_STRESS',
+                        operation='Get test_options')
+            cls.test_options = hw_manager.get_test_options()
+
+            # Get DUT object
+            hlsa_logger(level='debug', ident='Description', test_code='HL338_FW_UPDATE_STRESS', operation='Get DUT')
+            cls.device = hw_manager.get_dut_component()
+
+            # Get PCI machine (start SSH session)
+            hlsa_logger(level='debug', ident='Description', test_code='HL338_FW_UPDATE_STRESS', operation='Get PCI machine')
+            host = hw_manager.get_machine_component("pci_machine")
+            # Get cli control
+            cls.cli_control = host.get_control(hw_manager.controls.cli)
+
+            # Get device_cli
+            hlsa_logger(level='debug', ident='Description', test_code='HL338_FW_UPDATE_STRESS', operation='Get device_cli')
+            cls.device_cli = host.get_control(hw_manager.controls.device_cli)
+
+            # Get driver_cli
+            hlsa_logger(level='debug', ident='Description', test_code='HL338_FW_UPDATE_STRESS', operation='Get driver_cli')
+            cls.driver_cli = host.get_control(hw_manager.controls.driver_cli)
+
+            # Removing the driver before starting the test
+            status = cls.driver_cli.remove_driver()
+            hlsa_checker.check_status_is_true(status, ident='Description', test_code='HL338_FW_UPDATE_STRESS',
+                                              operation='Removing the driver before starting the test')
+
+
+            # Verify the type of the machine under test
+            if cls.device.spec.dut_name is "HL338":
+                hlsa_logger(level='info', ident='Description', test_code='HL338_FW_UPDATE_STRESS',
+                            operation='###### Device under test is HL338 ######')
+                # Get MACHX05 component, controls and spec
+                hlsa_logger(level='debug', ident='Description', test_code='HL338_FW_UPDATE_STRESS',
+                            operation='Get MACHX05 component, controls and spec')
+                cls.machx05 = cls.device.get_component(hw_manager.components.MACHX05)
+                cls.machx05_registers = cls.machx05.get_control(hw_manager.controls.usb_i2c_registers)
+                cls.machx05_regs_json = cls.machx05.spec
+
+                # TODO - Verify same components as in HL338
+                # Get I2C-to-SPI (SC18IS602B) component, control and flash (MX25U1633F) spec
+                hlsa_logger(level='debug', ident='Description', test_code='HL338_FW_UPDATE_STRESS',
+                            operation='Get i2c_2_spi components and spec')
+                cls.i2c_2_spi = cls.device.get_component(hw_manager.components.CPLD_I2C2SPI)
+                # Get flash memory (MX25U1633F) spec
+                cls.spi_spec = cls.device.get_component(hw_manager.components.MX25U1633F).spec
+
+                # Get FLASH_SPI_SEL (switch) initial value
+                cls._initial_FLASH_SPI_SEL_value = cls.machx05_registers.read_bit(
+                    cls.machx05_regs_json.configuration3_reg,
+                    cls.machx05_regs_json.configuration3_reg.bits.FLASH_SPI_SEL.index)
+                # Verify value is not None
+                hlsa_checker.check_value_is_not_none(cls._initial_FLASH_SPI_SEL_value, ident='Value',
+                                                     test_code='HL338_FW_UPDATE_STRESS',
+                                                     operation='FLASH_SPI_SEL value')
+
+                # Set up FLASH_SPI_SEL (in SPI flash switch) to point to the CPLD
+                written = cls.machx05_registers.write_bit(cls.machx05_regs_json.configuration3_reg,
+                                                          cls.machx05_regs_json.configuration3_reg.bits.FLASH_SPI_SEL.index,
+                                                          1)
+                hlsa_checker.check_status_is_true(written, ident='Description', test_code='HL338_FW_UPDATE_STRESS',
+                                                  operation='Write strap machx05.FLASH_SPI_SEL: 1',
+                                                  read_value=str(
+                                                      cls.machx05_regs_json.configuration3_reg.bits.FLASH_SPI_SEL.__name__) + ' 1')
+
+                # Read CPLD's SPI write protect boot strap (FlASH_SPI_0_WP#) original value
+                cls._initial_FLASH_SPI_0_WPn_value = cls.machx05_registers.read_bit(
+                    cls.machx05_regs_json.configuration3_reg,
+                    cls.machx05_regs_json.configuration3_reg.bits.FLASH_SPI_0_WP_N.index)
+                hlsa_checker.check_value_is_not_none(cls._initial_FLASH_SPI_0_WPn_value, ident='Description',
+                                                     test_code='HL338_FW_UPDATE_STRESS',
+                                                     operation='Read machx05 strap FLASH_SPI_0_WPn')
+
+                # Read Initial i2c_2_spi component status register value
+                cls._initial_status_reg_data = cls.i2c_2_spi.get(cls.spi_spec.instructions.READ_STATUS_REGISTER)
+                hlsa_checker.check_value_is_not_none(cls._initial_status_reg_data, ident='Description',
+                                                     test_code='HL338_FW_UPDATE_STRESS', operation='Set spi write enable')
+
+                # Set FLASH_SPI_0_WP# = 1 to DISABLE SPI Write Protect (To enable writing)
+                written = cls.machx05_registers.write_bit(cls.machx05_regs_json.configuration3_reg,
+                                                          cls.machx05_regs_json.configuration3_reg.bits.FLASH_SPI_0_WP_N.index,
+                                                          1)
+                hlsa_checker.check_status_is_true(written, ident='Description', test_code='HL338_FW_UPDATE_STRESS',
+                                                  operation='Write strap machx05.FLASH_SPI_0_WPn %s' %
+                                                            str(
+                                                                cls.machx05_regs_json.configuration3_reg.bits.FLASH_SPI_0_WP_N),
+                                                  read_value='1')
+
+                # Enable i2c_2_spi writing
+                status = cls.i2c_2_spi.set(cls.spi_spec.instructions.WRITE_ENABLE)
+                hlsa_checker.check_status_is_true(status, ident='Description', test_code='HL338_FW_UPDATE_STRESS',
+                                                  operation='i2c_2_spi.WRITE_ENABLE')
+
+            # pylint:disable=broad-except
+        except Exception as exception:
+            hlsa_logger(level='error', ident='Description', test_code='HL338_FW_UPDATE_STRESS',
+                        operation='Test Initialization Failure',
+                        read_value='Exception:' + str(exception), result='FAIL')
+            pytest.fail()

+        cmd = "/software/releases/%s/firmware/%s" % (self.test_options.fw_latest_ver, "gaudi3-release/gaudi3-agent-fw_loader-fit.itb")
+        burn_status = self.driver_cli.flash_firmware_file(cmd, source='host_pc')
+        hlsa_checker.check_status_is_true(burn_status, ident='Description', test_code='HL338_FW_UPDATE_STRESS',
+                                          operation='Test Initialization Failure',
+                                          read_value='Teardown burning firmware %s, gaudi3-release'
+                                                     % self.test_options.fw_latest_ver)
