+#!/bin/sh -eux
+
+VAGRANT_HOME="${HOME}/.vagrant.d"
+box_os="centos"
+box_name="${box_os}-7.5"
+# box_version="201808.24.0"
+box_version="20250114"
+box_url="https://app.vagrantup.com/bento/boxes/${box_name}/versions/${box_version}/providers/virtualbox.box"
+
+sudo rm -f box
+
+vagrant mutate "${box_url}" libvirt
+
+./templates/_common/create_box.sh "${VAGRANT_HOME}/boxes/virtualbox/${box_version}/libvirt/box.img" box
+
+"curl -H 'X-JFrog-Art-Api:${MY_JFROG_API_KEY}' -T box \"https://artifactory-kfs.habana-labs.com/artifactory/vagrant-hl-local/os/${box_os}/${box_name}-libvirt-${box_version}.box;box_name=${box_name}-libvirt;box_provider=libvirt;box_version=${box_version}\""
