+ *
+ * Copyright 2025 Intel, Ltd.

+#ifndef JGS_BOOT_DESC_H
+#define JGS_BOOT_DESC_H
+
+#include <stdint.h>
+
+#ifndef __packed
+#define __packed	__attribute__((__packed__))
+#endif
+
+// /* FW boot pointers magic:	  N X T 0 */
+
+#define HL_CHKSUM_SIZE		32
+#define HL_SHA_384_CSUM_SIZE	48
+
+struct fw_boot_magic {
+	union {
+		struct {
+			uint8_t type;	/* fw_img_type */
+			char name[3];
+		} __packed;
+		uint32_t val;
+	};
+};
+
+/*
+ *           struct fw_boot_ptrs                     struct fw_img_desc
+ *           =======================                       =================
+ *          || Next/Boot_ptrs_img_sz||              +---->|| Magic         ||
+ *          | ---------------------- |              |     | --------------- |
+ *          || Mod-able/Img_desc_sz ||              |     || Version       ||
+ *          | ---------------------- |              |     | --------------- |
+ *          || Magic                ||              |     || Index         ||
+ *          | ---------------------- |              |     | --------------- |
+ *          || Count                ||              |     || Offset    [2] ||
+ *          | ---------------------- |              |     | --------------- |
+ *          || Pub key sz           ||              |     || Size      [2] ||
+ *          | ---------------------- |              |     | --------------- |
+ *          || Sig sz               ||              |     || Load High [2] ||
+ *          | ---------------------- |              |     | --------------- |
+ *          || Sec ver              ||              |     || Load Low  [2] ||
+ *          | ---------------------- |              |     | --------------- |
+ *          || Key idx              ||              |     || SHA256    [2] ||
+ *          | ---------------------- |              |     | --------------- |
+ *          || Rsvd [2]             ||              |     || SHA384    [2] ||
+ *          | ---------------------- |              |     | --------------- |
+ *     +--<<|| Offset2array         ||              |     || Start High [2 ||
+ *     |      ======================                |     | --------------- |
+ *     |                                            |     || Start Low [2] ||
+ *     |                                            |      =================
+ *     |                                            |
+ *     |                                            |
+ *     |    Following Array                         |
+ *     |     ======================                 |
+ *     +--->|| Offset               ||>>------------+
+ *     |    | ---------------------- |                     =================
+ *     +--->|| Offset               ||>>----------------->|  ...            |
+ *     |    | ---------------------- |                     =================
+ *     :       :                         :                    :
+ *     |    | ---------------------- |                     =================
+ *     +--->|| Offset               ||>>----------------->|  ...            |
+ *           ========================                      =================
+ *
+ */
+
+struct fw_boot_ptrs {
+	union {
+		struct {
+			uint32_t next;	/* backward compatibility */
+			uint32_t mod;	/* backward compatibility */
+		};
+		struct {
+			/*
+			 * boot pointers image size
+			 * sizeof(((struct fw_boot_ptrs) + offsets_4B[count] +
+			 * (struct fw_img_desc[count])))
+			 */
+			uint32_t boot_ptrs_img_sz;
+			/*
+			 * single image descriptor size
+			 * sizeof(struct fw_img_desc)
+			 */
+			uint32_t img_desc_sz;
+		};
+	};
+	struct fw_boot_magic magic;
+	uint32_t count;
+	uint32_t pub_key_sz;	/* Public key size */
+	uint32_t sig_sz;	/* Signature size */
+	uint8_t sec_ver;	/* Security version of image */
+	uint8_t key_idx;	/* Key index of image */
+	uint8_t rsvd[2];	/* Reserved section for alignment */
+	uint32_t offset2array;	/* Offset to array of offsets */
+};
+
+struct fw_img_desc {
+	struct fw_boot_magic magic;
+	uint32_t desc_ver;	/* descriptor version */
+	uint32_t idx;		/* "next" index inside the offset array */
+	uint32_t offset[2];	/* the offset on the SPI flash */
+	uint32_t size[2];	/* the size of the image to load */
+	uint32_t load_high[2];	/* the high [32-63] bits of the load address */
+	uint32_t load_low[2];	/* the low [0-31] bits of the load address */
+	uint8_t chksum[2][HL_CHKSUM_SIZE];	/* sha256 checksum */
+	uint8_t chksum2[2][HL_SHA_384_CSUM_SIZE];	/* sha384 checksum */
+	uint32_t start_high[2];	/* the high [32-63] bits of the start address */
+	uint32_t start_low[2];	/* the low [0-31] bits of the start address */
+};
+
+#endif /* JGS_BOOT_DESC_H */
